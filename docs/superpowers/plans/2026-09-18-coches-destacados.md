# Coches Destacados Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let Andrés mark up to 3 cars per profile (his own / Alejandro's) as "destacados" from any device, with changes live on the public site within ~1 minute, with no local script run and no manual `git push`.

**Architecture:** A new static admin page (`/admin/index.html`) writes directly to `destacados.json` in the GitHub repo via the GitHub Contents API, using a personal access token stored in the browser's `localStorage`. The public catalog pages (`index.html`, `alejandro/index.html`, both generated by `generar_web.py`) fetch that file client-side on load and render a "Destacados" strip above the normal grid. A new `coches_lista.json` (generated daily by `generar_web.py`) feeds the admin page's car picker without exposing the full internal `datos_coches.json`.

**Tech Stack:** Python 3 (existing `generar_web.py` generator, no new dependencies), vanilla JS/CSS (existing site has none — stays that way), GitHub REST API (Contents endpoint) called directly from the browser.

**Spec:** `docs/superpowers/specs/2026-09-18-coches-destacados-design.md`

### Task 2: Make the photo-nav buttons work by event delegation (not per-node listeners)

**Files:**
- Modify: `generar_web.py` inside `build_index_html`, the `<script>` block that currently starts with `document.querySelectorAll('.rd-card-nav').forEach(...)`

The destacados strip (Task 5) will clone existing `.rd-card` nodes after the page has already loaded. Clones don't inherit JS event listeners that were attached per-node at load time, so their prev/next photo arrows would silently do nothing. Switching to a single delegated listener on `document` fixes this for both the original cards and any future clones, with no extra wiring needed per clone.

- [ ] **Step 1: Replace the per-node listener with delegation**

Find this block near the top of the `<script>` in `build_index_html`:
```python
<script>

  btn.addEventListener('click', e => {{
    e.preventDefault();
    e.stopPropagation();
    const media = btn.closest('.rd-card-media');
    const imgs  = [...media.querySelectorAll('.rd-card-photos img')];
    const dots  = [...media.querySelectorAll('.rd-card-dot')];
    let idx = imgs.findIndex(img => img.classList.contains('activa'));
    idx = btn.classList.contains('next')
      ? (idx + 1) % imgs.length
      : (idx - 1 + imgs.length) % imgs.length;
    imgs.forEach((img, i) => img.classList.toggle('activa', i === idx));
    dots.forEach((d, i) => d.classList.toggle('activa', i === idx));
  }});
}});
```

```python
def build_coches_lista_json(cars: list[dict], rutas: dict[int, list[str]]) -> str:
    """Lista pública resumida de coches disponibles — la usa /admin/ para
    mostrar fotos+modelo+precio al elegir destacados. No expone
    datos_coches.json completo (equipamiento, urls internas de DWA, etc.).
    Solo coches 'Disponible' — no tiene sentido destacar uno ya reservado."""
    items = []
    for car in cars:
        if car["estado"] != "Disponible":
            continue
        n = car["n"]
        fotos = (
            [f"/{f.lstrip('/')}" for f in car.get("fotos", [])]
            if car.get("fuente") == "motorflash" else rutas.get(n, [])
        )
        items.append({
            "n": n,
            "modelo": car["modelo"],
            "precio": car["precio"],
            "foto": fotos[0] if fotos else "",
        })
    return json.dumps(items, ensure_ascii=False, indent=2)
```

        renderLista();
      });
      perfilTabs.appendChild(btn);
    });
  }

  function cargarSeleccionDePerfil() {
    seleccion = {};
    (destacadosActuales[perfilActivo] || []).forEach(item => { seleccion[item.n] = item.etiqueta; });
  }

  function renderLista() {
    const seleccionados = Object.keys(seleccion).length;
    contador.textContent = `${seleccionados} / ${MAX_DESTACADOS} seleccionados`;
    listaCoches.innerHTML = '';
    cochesDisponibles.forEach(coche => {
      const marcado = coche.n in seleccion;
      const limiteAlcanzado = !marcado && seleccionados >= MAX_DESTACADOS;
      const row = document.createElement('div');
      row.className = 'coche-row';
      row.innerHTML = `
        <input type="checkbox" class="coche-check" ${marcado ? 'checked' : ''} ${limiteAlcanzado ? 'disabled' : ''}>
        <img src="${coche.foto}" alt="">
        <div class="coche-info">
          <div class="modelo">${coche.modelo}</div>
          <div class="precio">${coche.precio} €</div>
          <select ${marcado ? '' : 'hidden'}>
            ${FRASES.map(f => `<option value="${f}" ${seleccion[coche.n] === f ? 'selected' : ''}>${f}</option>`).join('')}
          </select>
        </div>`;
      const check = row.querySelector('.coche-check');
      const select = row.querySelector('select');
      check.addEventListener('change', () => {
        if (check.checked) {
          seleccion[coche.n] = select.value || FRASES[0];
        } else {
          delete seleccion[coche.n];
        }
        renderLista();
      });
      select.addEventListener('change', () => { seleccion[coche.n] = select.value; });
      listaCoches.appendChild(row);
    });
  }

  async function cargarTodo() {
    listaCoches.textContent = 'Cargando coches…';
    try {
      const [lista, dest] = await Promise.all([
        fetch('/coches_lista.json').then(r => r.json()),
        ghGet('destacados.json')
      ]);
      cochesDisponibles = lista;
      destacadosActuales = dest.data || {};
      destacadosSha = dest.sha;
      cargarSeleccionDePerfil();
      renderPerfilTabs();
      renderLista();
      btnGuardar.disabled = false;
      authCard.hidden = true;
      panel.hidden = false;
    } catch (e) {
      mostrarEstado('No se pudo cargar: revisa el token. (' + e.message + ')', true);
      authCard.hidden = false;
      panel.hidden = true;
    }
  }

  document.getElementById('btn-guardar-token').addEventListener('click', () => {
    const t = tokenInput.value.trim();
    if (!t) return;
    localStorage.setItem(TOKEN_KEY, t);
    cargarTodo();
  });

  document.getElementById('btn-cambiar-token').addEventListener('click', () => {
    localStorage.removeItem(TOKEN_KEY);
    tokenInput.value = '';
    panel.hidden = true;
    authCard.hidden = false;
  });

  document.getElementById('btn-guardar').addEventListener('click', async () => {
    btnGuardar.disabled = true;
    mostrarEstado('Guardando…', false);
    try {
      const lista = Object.keys(seleccion).map(n => ({ n: Number(n), etiqueta: seleccion[n] }));
      const nuevo = Object.assign({}, destacadosActuales, { [perfilActivo]: lista });
      const fecha = new Date().toISOString().slice(0, 10);
      const resultado = await ghPut('destacados.json', nuevo, destacadosSha, `Actualizar destacados (${perfilActivo}) — ${fecha}`);
      destacadosActuales = nuevo;
      destacadosSha = resultado.content.sha;
      mostrarEstado('Guardado. Puede tardar hasta un minuto en verse en la web.', false);
    } catch (e) {
      mostrarEstado('No se pudo guardar: ' + e.message, true);
    }
    btnGuardar.disabled = false;
  });

  if (getToken()) cargarTodo();
})();
</script>
</body>
</html>
